local CORRECT_KEY = "HuyPham"

local SAVE_FILE_NAME = "ChilliHubV5_KeyData.json"

-- Hàm tự động chạy script chính
local function loadMainScript()
	local success, err = pcall(function()
		loadstring(game:HttpGet("https://raw.githubusercontent.com/realmakehub-cell/leakti/refs/heads/main/main.lua"))()
	end)
	if not success then
		warn("Failed to load script: " .. tostring(err))
	end
end

-- ----------------------------------------------------
-- KIỂM TRA LƯU KEY (24 GIỜ)
-- ----------------------------------------------------
local function checkSavedKey()
	if readfile and isfile and isfile(SAVE_FILE_NAME) then
		local success, data = pcall(function()
			return HttpService:JSONDecode(readfile(SAVE_FILE_NAME))
		end)
		if success and type(data) == "table" then
			if data.Key == CORRECT_KEY and data.ExpireTime and os.time() < data.ExpireTime then
				return true
			end
		end
	end
	return false
end

local function saveKeyData()
	if writefile then
		local data = {
			Key = CORRECT_KEY,
			ExpireTime = os.time() + (24 * 3600) -- Hết hạn sau 24 giờ (86400 giây)
		}
		pcall(function()
			writefile(SAVE_FILE_NAME, HttpService:JSONEncode(data))
		end)
	end
end

-- Nếu đã lưu key và chưa hết hạn 24 giờ -> Chạy thẳng script chính, không hiện Menu
if checkSavedKey() then
	loadMainScript()
	return
end

-- Function to trigger in-menu alert
local notifThread = nil
local function setInMenuAlert(msg, isSuccess)
	AlertText.Text = msg
	
	if isSuccess then
		AlertFrame.BackgroundColor3 = Color3.fromRGB(18, 32, 22)
		AlertStroke.Color = Color3.fromRGB(50, 255, 120)
		AlertText.TextColor3 = Color3.fromRGB(80, 255, 140)
	else
		AlertFrame.BackgroundColor3 = Color3.fromRGB(32, 18, 18)
		AlertStroke.Color = Color3.fromRGB(255, 50, 50)
		AlertText.TextColor3 = Color3.fromRGB(255, 80, 80)
	end

	TweenService:Create(AlertFrame, TweenInfo.new(0.25), {BackgroundTransparency = 0.2}):Play()
	TweenService:Create(AlertStroke, TweenInfo.new(0.25), {Transparency = 0.3}):Play()
	TweenService:Create(AlertText, TweenInfo.new(0.25), {TextTransparency = 0}):Play()

	if notifThread then task.cancel(notifThread) end
	notifThread = task.delay(2.5, function()
		TweenService:Create(AlertFrame, TweenInfo.new(0.3), {BackgroundTransparency = 1}):Play()
		TweenService:Create(AlertStroke, TweenInfo.new(0.3), {Transparency = 1}):Play()
		TweenService:Create(AlertText, TweenInfo.new(0.3), {TextTransparency = 1}):Play()
	end)
end

-- Ultra Smooth Math Sine Elastic Shake
local function ultraSmoothShake()
	local basePos = UDim2.new(0.5, -180, 0.5, -120)
	
	TweenService:Create(UIStroke, TweenInfo.new(0.12), {Color = Color3.fromRGB(255, 0, 0), Transparency = 0}):Play()
	
	task.spawn(function()
		local startTime = tick()
		local duration = 0.4
		while tick() - startTime < duration do
			local elapsed = tick() - startTime
			local progress = elapsed / duration
			local offset = math.sin(elapsed * 50) * 9 * (1 - progress)
			MainFrame.Position = basePos + UDim2.new(0, offset, 0, 0)
			task.wait()
		end
		MainFrame.Position = basePos
		TweenService:Create(UIStroke, TweenInfo.new(0.35), {Color = Color3.fromRGB(255, 35, 35), Transparency = 0.2}):Play()
	end)
end

-- Submit Button Event
SubmitBtn.MouseButton1Click:Connect(function()
	if TextBox.Text == CORRECT_KEY then
		saveKeyData() -- Lưu Key vào file (có hạn 24 giờ)
		setInMenuAlert("Valid Key! Loading script...", true)
		
		local tween = TweenService:Create(MainFrame, TweenInfo.new(0.35, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
			Size = UDim2.new(0, 0, 0, 0),
			Position = UDim2.new(0.5, 0, 0.5, 0)
		})
		tween:Play()
		task.wait(0.35)
		ScreenGui:Destroy()
		
		loadMainScript()
	else
		TextBox.Text = ""
		ultraSmoothShake()
		setInMenuAlert("Invalid Key! Please try again.", false)
	end
end)
